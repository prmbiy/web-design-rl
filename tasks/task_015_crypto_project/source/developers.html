$ npm install @veltrax/sdk

// Initialize Veltrax client
import { VeltraxClient, Wallet } from '@veltrax/sdk';

const client = new VeltraxClient({
  rpc: 'https://mainnet.veltrax.io',
  chainId: 8453,
});

const wallet = Wallet.fromKey(process.env.PRIVATE_KEY);

// Deploy contract in parallel slot
const tx = await client.deployContract({
  bytecode,
  abi,
  signer: wallet,
});

console.log(`Deployed at ${tx.address} · ${tx.finalityMs}ms`);
