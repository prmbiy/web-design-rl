// Evaluate a feature flag at the edge
import { Flexor } from '@flexor/node';

const flexor = new Flexor({
  apiKey: process.env.FLEXOR_KEY,
  edge: true
});

const user = {
  id: 'usr_8f2a',
  plan: 'pro',
  region: 'us-east'
};

  user
);

  renderNewCheckout();
}
